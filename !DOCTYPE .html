ตั้งค่าการเชื่อมต่อ Google Sheets

กรอก URL ของ Google Apps Script (Web App) ของคุณ เพื่อส่งข้อมูลจากแบบฟอร์มนี้ตรงเข้า Google Sheet ทันที

*หากไม่ระบุ ระบบจะใช้โหมดจำลองการบันทึก (Mock Mode) ซึ่งสามารถทดสอบกดส่งได้จริง

📋 วิธีรับ URL ไปยัง Google Sheets ใน 2 นาที:

  1. เปิด Google Sheets ใหม่ และสร้างหัวตารางในแถวที่ 1:
    Timestamp | ชื่อ-นามสกุล | เบอร์โทร | อีเมล | รุ่นแอร์ | จุดเด่นที่ชื่นชอบ | คะแนน | ความคิดเห็นเพิ่มเติม
  2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
  3. วางโค้ดด้านล่างนี้ลงไป แล้วคลิก Deploy (ทำให้ใช้งานได้) > New deployment > เลือก Web app > ตั้งค่า Who has access: Anyone แล้วคัดลอก URL มาใส่ในช่องด้านบน
function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    sheet.appendRow([
      new Date(),
      data.fullName,
      data.phoneNumber,
      data.email,
      data.carrierModel,
      data.favoriteFeatures,
      data.satisfactionScore,
      data.userFeedback
    ]);
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
