เชื่อมต่อ Google Sheets

บันทึกข้อมูลทุกคำตอบลงตาราง Google Sheet ของคุณแบบ Real-time

ขณะนี้อยู่ในโหมดจำลอง (ยังไม่ได้ใส่ Web App URL)

🛠️ วิธีสร้างลิงก์เชื่อมต่อ Google Sheet ใน 4 ขั้นตอนง่ายๆ:

  1. สร้าง Sheet ใหม่: ใส่หัวตารางแถวที่ 1 (คอลัมน์ A ถึง H) ดังนี้:
    วัน-เวลาที่ส่ง | 1. ชื่อ - นามสกุล | 2. เบอร์โทรศัพท์ | 3. อีเมล | 4. คุณใช้แอร์แคเรียร์รุ่นอะไร | 5. ชื่นชอบอะไรในแคเรียร์ | ระดับความพึงพอใจ (1-5 ดาว) | ข้อเสนอแนะเพิ่มเติม
  2. ไปที่เมนู ส่วนขยาย (Extensions) → Apps Script
  3. ลบโค้ดเดิมทิ้ง แล้ววางโค้ดด้านล่างนี้ลงไป จากนั้นกด บันทึก (Ctrl+S)
  4. คลิกปุ่ม ทำให้ใช้งานได้ (Deploy) → การทำให้ใช้งานได้รายการใหม่ (New deployment):
    • เลือกประเภท: เว็บแอป (Web app)
    • ผู้มีสิทธิ์เข้าถึง (Who has access): เลือก ทุกคน (Anyone)
    • กด ทำให้ใช้งานได้ แล้วคัดลอก URL เว็บแอป มาวางในช่องด้านบน
function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000); // ป้องกันข้อมูลชนกันเมื่อมีการส่งพร้อมกัน
  
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      new Date(),                 // วันเวลา
      data.fullName || "",         // 1. ชื่อ-นามสกุล
      data.phoneNumber || "",      // 2. เบอร์โทร
      data.email || "",            // 3. Email
      data.carrierModel || "",      // 4. รุ่นแอร์
      data.favoriteFeatures || "", // 5. จุดเด่นที่ชื่นชอบ
      data.satisfactionScore || "",// คะแนน (1-5)
      data.userFeedback || ""      // ข้อเสนอแนะเพิ่มเติม
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", error: err.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
